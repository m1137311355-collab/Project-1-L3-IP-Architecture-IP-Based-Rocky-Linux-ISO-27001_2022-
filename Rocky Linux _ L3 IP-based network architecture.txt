Stratis File System ( Server & Client ) 
==============================================================================================================================================
                        
# Server
	sudo dnf install nfs-utils -y  
	
	sudo systemctl enable --now nfs-server
	sudo exportfs -r
	sudo firewall-cmd --add-service=nfs --permanent
	sudo firewall-cmd --permanent --add-service=rpc-bind
	sudo firewall-cmd --permanent --add-service=mountd
	sudo firewall-cmd --reload
	
	sudo systemctl enable --now rpcbind
	sudo systemctl restart nfs-server
	
	mkdir -p /srv/stratis_share
	chmod 777 /srv/stratis_share
	sudo chcon -t public_content_rw_t /srv/stratis_share
	sudo setsebool -P nfs_export_all_rw 1
	
	sudo setenforce 0
	sudo setsebool -P nfs_export_all_rw 1
	sudo setsebool -P nfs_export_all_ro 1
	sudo restorecon -R /srv/stratis_share
	sudo setenforce 1
	
	
	vim /etc/exports
		/srv/stratis_share 192.168.213.0/24(rw,sync,no_root_squash,insecure))
	sudo exportfs -rv
	systemctl deamon-reload


# Client 
	sudo dnf install nfs-utils -y   
	sudo mkdir -p /mnt/client_stratis
	sudo mount -t nfs 192.168.213.14:/srv/stratis_share /mnt/client_stratis
	vim /etc/fstab
		192.168.213.14:/srv/stratis_share  /mnt/client_stratis  nfs  defaults,_netdev  0 0
